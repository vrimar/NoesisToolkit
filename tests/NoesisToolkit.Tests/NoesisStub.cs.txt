// A compile-only stand-in for the Noesis surface the generated code touches. Only the shape matters:
// the tests assert that emitted C# binds and compiles, never that it renders.

public class DependencyProperty
{
    public static readonly object UnsetValue = new object();

    public string Name = "";

    public static DependencyProperty Register(string name, System.Type propertyType, System.Type ownerType, PropertyMetadata metadata) => new DependencyProperty { Name = name };

    public static DependencyProperty RegisterAttached(string name, System.Type propertyType, System.Type ownerType, PropertyMetadata metadata) => new DependencyProperty { Name = name };
}

public class PropertyMetadata

    public PropertyMetadata(object? defaultValue) { }

    public PropertyMetadata(object? defaultValue, PropertyChangedCallback? callback) { }
}

public class FrameworkPropertyMetadata : PropertyMetadata
{
    public FrameworkPropertyMetadata() { }

    public FrameworkPropertyMetadata(object? defaultValue) : base(defaultValue) { }

    public FrameworkPropertyMetadata(object? defaultValue, FrameworkPropertyMetadataOptions options) : base(defaultValue) { }

    public FrameworkPropertyMetadata(object? defaultValue, FrameworkPropertyMetadataOptions options, PropertyChangedCallback? callback) : base(defaultValue, callback) { }

    public FrameworkPropertyMetadata(object? defaultValue, PropertyChangedCallback? callback) : base(defaultValue, callback) { }
}

[System.Flags]
public enum FrameworkPropertyMetadataOptions
{
    None = 0,
    AffectsMeasure = 1,
    AffectsArrange = 2,
    AffectsRender = 4,
    BindsTwoWayByDefault = 8,
}

public delegate void PropertyChangedCallback(DependencyObject d, DependencyPropertyChangedEventArgs e);

public class DependencyPropertyChangedEventArgs : System.EventArgs
{
    public object? OldValue;
    public object? NewValue;
}

public class DependencyObject
{
    readonly System.Collections.Generic.Dictionary<DependencyProperty, object?> _values = new();

    public object? GetValue(DependencyProperty dp) => _values.TryGetValue(dp, out var v) ? v : null;

    public void SetValue(DependencyProperty dp, object? value) => _values[dp] = value;

    public void ClearValue(DependencyProperty dp) => _values.Remove(dp);

    public void SetResourceReference(DependencyProperty dp, object key) { }
}

public class FrameworkElement : DependencyObject
{
    public BindingExpressionBase? SetBinding(DependencyProperty dp, BindingBase binding) => null;

    public static readonly DependencyProperty WidthProperty = DependencyProperty.Register("Width", typeof(float), typeof(FrameworkElement), new FrameworkPropertyMetadata());
    public static readonly DependencyProperty HeightProperty = DependencyProperty.Register("Height", typeof(float), typeof(FrameworkElement), new FrameworkPropertyMetadata());

    public static readonly DependencyProperty VisibilityProperty = DependencyProperty.Register("Visibility", typeof(Visibility), typeof(FrameworkElement), new FrameworkPropertyMetadata());

    public string? Name { get; set; }

    public Style? Style { get; set; }

    public Visibility Visibility { get => (Visibility)(GetValue(VisibilityProperty) ?? Visibility.Visible); set => SetValue(VisibilityProperty, value); }

    public float Width { get => (float)(GetValue(WidthProperty) ?? 0f); set => SetValue(WidthProperty, value); }

    public float Height { get => (float)(GetValue(HeightProperty) ?? 0f); set => SetValue(HeightProperty, value); }

    public ResourceDictionary Resources { get; set; } = new ResourceDictionary();

    public InputBindingCollection InputBindings { get; } = new InputBindingCollection();

    public static readonly DependencyProperty DataContextProperty = DependencyProperty.Register("DataContext", typeof(object), typeof(FrameworkElement), new FrameworkPropertyMetadata());

    public object? DataContext { get => GetValue(DataContextProperty); set => SetValue(DataContextProperty, value); }

    public static readonly DependencyProperty TagProperty = DependencyProperty.Register("Tag", typeof(object), typeof(FrameworkElement), new FrameworkPropertyMetadata());

    public object? Tag { get => GetValue(TagProperty); set => SetValue(TagProperty, value); }

    public object? FindName(string name) => null;

    public event System.EventHandler? Loaded;

    protected virtual bool ConnectEvent(object source, string eventName, string handlerName) => false;

    protected void RaiseLoaded() => Loaded?.Invoke(this, System.EventArgs.Empty);
}

public class ResourceDictionary
{
    readonly System.Collections.Generic.Dictionary<object, object?> _entries = new();

    public System.Collections.Generic.List<ResourceDictionary> MergedDictionaries { get; } = new();

    public object? Source { get; set; }

    public System.Collections.ICollection Keys => _entries.Keys;

    public bool Contains(object key) => _entries.ContainsKey(key);

    public object? this[object key]
    {
        get => _entries.TryGetValue(key, out var v) ? v : null;
        set => _entries[key] = value;
    }

    public void Add(object key, object? value) => _entries[key] = value;
}
