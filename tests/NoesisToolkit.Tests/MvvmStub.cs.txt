// The runtime surface the MVVM generators emit against, mirrored so the tests need no package
// reference back to NoesisToolkit.Mvvm.
namespace NoesisToolkit.Mvvm
{
    [System.AttributeUsage(System.AttributeTargets.Property)]
    public sealed class DependencyPropertyAttribute : System.Attribute
    {
        public DependencyPropertyAttribute() { }

        public DependencyPropertyAttribute(object? defaultValue, string? propertyChanged = null, Noesis.FrameworkPropertyMetadataOptions metadataOptions = Noesis.FrameworkPropertyMetadataOptions.None) { }
    }

    [System.AttributeUsage(System.AttributeTargets.Method)]
    public sealed class DelegateCommandAttribute : System.Attribute { }

    public class DelegateCommand<T>
    {
        public DelegateCommand(System.Action<T?> execute) { }
    }

    public sealed class DelegateCommand : DelegateCommand<object?>
    {
        public DelegateCommand(System.Action execute) : base(_ => execute()) { }
    }

    public class AsyncDelegateCommand<T>
    {
        public AsyncDelegateCommand(System.Func<T?, System.Threading.Tasks.ValueTask> execute) { }
    }

    public sealed class AsyncDelegateCommand : AsyncDelegateCommand<object?>
    {
        public AsyncDelegateCommand(System.Func<System.Threading.Tasks.ValueTask> execute) : base(_ => execute()) { }
    }
}

namespace NoesisToolkit.Mvvm.CodeGen
{
    public static class DependencyWatcher
    {
        public static Noesis.DependencyProperty SelfNotifying(Noesis.DependencyProperty property) => property;

        public static Noesis.PropertyChangedCallback Notifying(Noesis.PropertyChangedCallback? inner) => static (_, _) => { };
        public static Noesis.FrameworkPropertyMetadata Metadata(object? defaultValue, Noesis.FrameworkPropertyMetadataOptions options, Noesis.PropertyChangedCallback? inner) => new Noesis.FrameworkPropertyMetadata(defaultValue, options, inner);
    }

    public readonly struct BindingHop
    {
        public BindingHop(string name, System.Func<object, object?> read) { }

        public static BindingHop Guarded<TOwner>(string name, System.Func<TOwner, object?> read) => default;

        public static BindingHop Bool(string name, System.Func<object, bool> read) => default;

        public static BindingHop GuardedBool<TOwner>(string name, System.Func<TOwner, bool> read) => default;

        public static BindingHop Value<TValue>(string name, System.Func<object, TValue> read) where TValue : struct => default;

        public static BindingHop GuardedValue<TOwner, TValue>(string name, System.Func<TOwner, TValue> read) where TValue : struct => default;
    }

    public static class DependencyRead
    {
        public static object? Value(Noesis.DependencyObject source, Noesis.DependencyProperty property) => null;
        public static bool Bool(Noesis.DependencyObject source, Noesis.DependencyProperty property) => false;
        public static int Int(Noesis.DependencyObject source, Noesis.DependencyProperty property) => 0;
        public static long Long(Noesis.DependencyObject source, Noesis.DependencyProperty property) => 0L;
        public static float Float(Noesis.DependencyObject source, Noesis.DependencyProperty property) => 0f;
        public static double Double(Noesis.DependencyObject source, Noesis.DependencyProperty property) => 0d;
        public static string String(Noesis.DependencyObject source, Noesis.DependencyProperty property) => "";
        public static TEnum Enum<TEnum>(Noesis.DependencyObject source, Noesis.DependencyProperty property) where TEnum : struct, System.Enum => default;
    }

    public static class ImageSources
    {
        public static Noesis.ImageSource From(string uri) => null!;
    }

    public static class DependencyWrite
    {
        public static void Value<T>(Noesis.DependencyObject target, Noesis.DependencyProperty property, T value) { }
    }

    public static class CompiledBindingSetup
    {
        public static int Register(string key, System.Action<Noesis.FrameworkElement> wire) => 0;

        public static void SetIndex(Noesis.DependencyObject target, int value) { }
    }

    public abstract class BindingLane
    {
        public static BindingLane Of<TOwner, TSource, TSlot>(System.Func<TOwner, TSource> read, System.Func<TSource, TSlot> convert, System.Action<TOwner, TSlot>? writeBack, bool guarded) where TSource : struct where TSlot : struct => null!;

        public static BindingLane Text<TOwner, TSource>(System.Func<TOwner, TSource> read, SlotFormat<TSource> format, bool guarded) where TSource : unmanaged => null!;
    }

    public delegate bool SlotFormat<T>(T value, System.Span<char> destination, out int written);

    public ref struct SlotText
    {
        public SlotText(System.Span<char> destination) { }

        public void Literal(string text) { }

        public void Integer<T>(T value) where T : struct, System.ISpanFormattable { }

        public void Integer<T>(T value, string format) where T : struct, System.ISpanFormattable { }

        public void Number(double value) { }

        public void Number(float value) { }

        public void Fixed(double value, string format) { }

        public void Fixed(float value, string format) { }

        public readonly bool Done(out int written) { written = 0; return true; }
    }

    public sealed class CompiledBindingSpec
    {
        public System.Func<Noesis.FrameworkElement, Noesis.FrameworkElement?>? Source { get; set; }
        public Noesis.DependencyProperty? SourceProperty { get; set; }
        public BindingHop[] Hops { get; set; } = System.Array.Empty<BindingHop>();
        public System.Func<object?, object?> Convert { get; set; } = static v => v;
        public System.Action<Noesis.FrameworkElement, object?>? Assign { get; set; }
        public System.Action<object, object?>? Write { get; set; }
        public BindingLane? Lane { get; set; }
        public Noesis.IValueConverter? Converter { get; set; }
        public object? ConverterParameter { get; set; }
        public System.Type? TargetType { get; set; }
        public Noesis.BindingMode Mode { get; set; } = Noesis.BindingMode.Default;
        public Noesis.UpdateSourceTrigger Trigger { get; set; } = Noesis.UpdateSourceTrigger.Default;
    }

    public sealed class CompiledBinding
    {
        public static CompiledBinding Bind(
            Noesis.FrameworkElement target,

            CompiledBindingSpec spec) => new CompiledBinding();

        public static CompiledBinding Bind(
            Noesis.FrameworkElement anchor,
            System.Func<Noesis.FrameworkElement, Noesis.DependencyObject?> receiver,
            Noesis.DependencyProperty property,
            CompiledBindingSpec spec) => new CompiledBinding();

        public static Noesis.FrameworkElement? FindAncestor(Noesis.FrameworkElement element, System.Type type) => null;

        public static Noesis.FrameworkElement? FindAncestorOrSelf(Noesis.FrameworkElement element, System.Type type) => null;

        public static Noesis.FrameworkElement? FindNamed(Noesis.FrameworkElement element, string name) => null;

        public static void MarkReceiver(Noesis.DependencyObject receiver, string key) { }

        public static Noesis.DependencyObject? MarkedBehavior(Noesis.FrameworkElement element, string key) => null;

        public static Noesis.DependencyObject? MarkedInputBinding(Noesis.FrameworkElement element, string key) => null;

        public static Noesis.DependencyObject? MarkedAction(Noesis.FrameworkElement element, string key) => null;
    }

    public static class SlotConversion

        public static string Text(float value, string format) => "";

        public static System.Func<object?, object?> Cached<TSource>(System.Func<TSource, object?> convert, object? fallback) where TSource : struct => static v => v;

    public sealed class CompiledTriggerCondition

    public sealed class CompiledSetter
    {
